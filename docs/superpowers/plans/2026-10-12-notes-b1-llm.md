> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Prove that llama.cpp runs Gemma 4 well enough for the notes job, then build the parts that the notes job (plan B2) needs: the `llm` module, the v3 schema with the notes stores, and job progress in the scheduler.

**Architecture:** Task 2 is a spike. It measures llama-cpp-python with Gemma 4 and writes decision 0003; if Gemma 4 does not load, the work stops. `forenote/llm/` has a `TextGenerator` protocol, a fake for tests, a llama.cpp backend that generates in a worker thread, and a `collect()` helper with the 120 s no-token watchdog. Migration v3 rebuilds `bullets` and `meeting_outputs` and adds `transcript_fts`. The scheduler passes a `report(message, fraction)` callback to each job.

**Tech Stack:** Python 3.12, asyncio, threads, llama-cpp-python 0.3.36 (built from source with CMake), SQLite FTS5, pytest.

- llama-cpp-python on PyPI is a source package. `uv add` builds it with CMake 3.28 and g++ (both installed). The build takes several minutes.
- llama-cpp-python API used in this plan:
  - `llama_cpp.Llama(model_path=..., n_ctx=..., n_threads=..., n_threads_batch=..., n_batch=512, verbose=False)`
  - `llm.metadata` is a dict of GGUF metadata strings. `"tokenizer.chat_template"` holds the Jinja chat template.
  - `llm.token_eos()`, `llm.token_bos()`, and `llm._model.token_get_text(token_id)` give the special token texts.
  - `llama_cpp.llama_chat_format.Jinja2ChatFormatter(template=..., eos_token=..., bos_token=...)`. Calling it with `messages=[{"role": "user", "content": text}]` returns an object with `.prompt`.
  - `llm.tokenize(data: bytes, add_bos: bool, special: bool) -> list[int]`.
  - `llm.create_completion(prompt_tokens, max_tokens=..., temperature=..., stream=True, grammar=...)` yields chunks; the text is `chunk["choices"][0]["text"]`. When the new prompt starts with the tokens of the last prompt, llama-cpp-python evaluates only the new tokens (prefix reuse).
  - `llama_cpp.LlamaGrammar.from_json_schema(json_text, verbose=False)`.
- The rendered chat prompt already starts with the BOS token text. Tokenize it with `add_bos=False, special=True`, or the model gets two BOS tokens.
- AMI meeting `ES2010c` is 30.5 minutes long with 5,357 words. Its manual word annotations are in `https://groups.inf.ed.ac.uk/ami/AMICorpusAnnotations/ami_public_manual_1.6.2.zip` (23 MB, CC BY 4.0). `tests/fixtures/ami/make_fixtures.py` already reads this zip.

```python
"""Build full meeting transcripts from the AMI word annotations. Needs network.

Run once and commit the output:
    uv run python tests/fixtures/ami/make_transcripts.py
"""

from __future__ import annotations

import html
import io
import json
import urllib.request
import xml.etree.ElementTree as ET
import zipfile
from pathlib import Path

HERE = Path(__file__).resolve().parent
WORDS_URL = "https://groups.inf.ed.ac.uk/ami/AMICorpusAnnotations/ami_public_manual_1.6.2.zip"
MEETINGS = ["ES2010c"]
ME = "A"  # the AMI speaker that plays "Me"; all other speakers are "Others"
GAP_S = 1.0  # a pause longer than this starts a new line
MAX_LINE_S = 20.0  # same limit as the live segmenter

LICENSE_EXTRA = """
The files in transcripts/ are full meeting transcripts built from the AMI
manual word annotations (ami_public_manual_1.6.2). Words of each speaker are
joined into lines at pauses longer than 1 s or after 20 s. Speaker A is
labelled "me", the other speakers "others".
"""


def words(z: zipfile.ZipFile, meeting: str) -> list[tuple[float, float, str, str]]:
    out = []
    for name in z.namelist():
        if not (name.startswith(f"words/{meeting}.") and name.endswith(".words.xml")):
            continue
        speaker = name.split(".")[1]
        for w in ET.fromstring(z.read(name)).iter("w"):
            if w.get("punc") == "true" or w.get("starttime") is None or w.get("endtime") is None:
                continue
            text = html.unescape((w.text or "").strip())
            if text:
                out.append((float(w.get("starttime")), float(w.get("endtime")), speaker, text))
    out.sort(key=lambda x: x[0])
    return out


def lines(ws: list[tuple[float, float, str, str]]) -> list[dict]:
    """Group the words of each speaker into lines, then sort all lines by start time."""
    by_speaker: dict[str, list[dict]] = {}
    for start, end, speaker, text in ws:
        cur = by_speaker.setdefault(speaker, [])
        last = cur[-1] if cur else None
        if last and start - last["end"] <= GAP_S and end - last["start"] <= MAX_LINE_S:
            last["words"].append(text)
            last["end"] = end
        else:
            cur.append({"start": start, "end": end, "speaker": speaker, "words": [text]})
    merged = sorted((ln for cur in by_speaker.values() for ln in cur), key=lambda ln: ln["start"])
    return [
        {
            "t_start_ms": round(ln["start"] * 1000),
            "t_end_ms": round(ln["end"] * 1000),
            "speaker": "me" if ln["speaker"] == ME else "others",
            "text": " ".join(ln["words"]),
        }
        for ln in merged
    ]


def main() -> None:
    with urllib.request.urlopen(WORDS_URL, timeout=120) as resp:
        z = zipfile.ZipFile(io.BytesIO(resp.read()))
    out_dir = HERE / "transcripts"
    out_dir.mkdir(exist_ok=True)
    for meeting in MEETINGS:
        rows = lines(words(z, meeting))
        with open(out_dir / f"{meeting}.jsonl", "w", encoding="utf-8") as f:
            for row in rows:
                f.write(json.dumps(row) + "\n")
        print(f"{meeting}: {len(rows)} lines, {rows[-1]['t_end_ms'] / 60000:.1f} min")
    lic = HERE / "LICENSE"
    text = lic.read_text(encoding="utf-8")
    if LICENSE_EXTRA.strip() not in text:
        lic.write_text(text + LICENSE_EXTRA, encoding="utf-8")


if __name__ == "__main__":
    main()
```

`spikes/llm/run_llm.py`:

```python
"""Spike: llama-cpp-python with Gemma 4 (notes spec, section 6.3).

Usage: uv run python spikes/llm/run_llm.py MODEL.gguf [--ctx N] [--threads N]
Prints one JSON object per check.
"""

from __future__ import annotations

import argparse
import json
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
TRANSCRIPT = ROOT / "tests" / "fixtures" / "ami" / "transcripts" / "ES2010c.jsonl"
TEMPLATE_MARKS = ("<start_of_turn>", "<end_of_turn>", "<|turn", "<turn|>")
ACTION_SCHEMA = {
    "type": "array",
    "items": {
        "type": "object",
        "properties": {"owner": {"type": ["string", "null"]}, "text": {"type": "string"}},
        "required": ["owner", "text"],
    },
}


def rss_mib() -> float:
    with open("/proc/self/status") as f:
        for line in f:
            if line.startswith("VmRSS:"):
                return int(line.split()[1]) / 1024
    return 0.0


def transcript_text(max_minutes: float) -> str:
    out = []
    with open(TRANSCRIPT, encoding="utf-8") as f:
        for raw in f:
            row = json.loads(raw)
            if row["t_start_ms"] > max_minutes * 60_000:
                break
            s = row["t_start_ms"] // 1000
            who = "Me" if row["speaker"] == "me" else "Others"
            out.append(f"[{s // 60:02d}:{s % 60:02d}] {who}: {row['text']}")
    return "\n".join(out)


def emit(**kw) -> None:
    print(json.dumps(kw), flush=True)


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("model", type=Path)
    ap.add_argument("--ctx", type=int, default=16384)
    ap.add_argument("--threads", type=int, default=7)
    ap.add_argument("--minutes", type=float, default=30.0)
    a = ap.parse_args()

    import llama_cpp
    from llama_cpp.llama_chat_format import Jinja2ChatFormatter

    before = rss_mib()
    t0 = time.perf_counter()
    llm = llama_cpp.Llama(
        model_path=str(a.model), n_ctx=a.ctx, n_threads=a.threads,
        n_threads_batch=a.threads, n_batch=512, verbose=False,
    )
    template = llm.metadata.get("tokenizer.chat_template")
    emit(check="load", model=a.model.name, llama_cpp=llama_cpp.__version__,
         arch=llm.metadata.get("general.architecture"), ctx=a.ctx, threads=a.threads,
         load_s=round(time.perf_counter() - t0, 1), rss_mib=round(rss_mib() - before),
         has_template=template is not None)
    if template is None:
        return 1

    bos = llm._model.token_get_text(llm.token_bos())
    eos = llm._model.token_get_text(llm.token_eos())
    fmt = Jinja2ChatFormatter(template=template, eos_token=eos, bos_token=bos)

    def render(text: str) -> str:
        return fmt(messages=[{"role": "user", "content": text}]).prompt

    sample = render("XYZ")
    emit(check="template", head=sample[: sample.index("XYZ")], tail=sample[sample.index("XYZ") + 3 :])

    prefix = (
        "You write meeting notes. Use only facts from the transcript.\n\n"
        f"Transcript:\n{transcript_text(a.minutes)}\n\n"
    )

    def run(task: str, max_tokens: int, grammar=None) -> dict:
        tokens = llm.tokenize(render(prefix + task).encode("utf-8"), add_bos=False, special=True)
        t = time.perf_counter()
        first = None
        n = 0
        text = ""
        for chunk in llm.create_completion(tokens, max_tokens=max_tokens, temperature=0.2,
                                           stream=True, grammar=grammar):
            if first is None:
                first = time.perf_counter() - t
            n += 1
            text += chunk["choices"][0]["text"]
        total = time.perf_counter() - t
        first = first if first is not None else total
        return {
            "prompt_tokens": len(tokens),
            "first_token_s": round(first, 2),
            "gen_tokens": n,
            "gen_tok_s": round((n - 1) / (total - first), 1) if n > 1 and total > first else 0.0,
            "stopped_early": n < max_tokens,
            "template_text_in_output": any(m in text for m in TEMPLATE_MARKS),
            "text": text,
        }

    r1 = run("Write a summary of the meeting in 3 to 6 sentences.", 500)
    r1["prompt_tok_s"] = round(r1["prompt_tokens"] / r1["first_token_s"], 1)
    emit(check="first_task", **r1)
    r2 = run("Expand this bullet with facts from the transcript in 1 to 3 sentences: «remote control buttons».", 200)
    emit(check="second_task_same_prefix", **r2)
    grammar = llama_cpp.LlamaGrammar.from_json_schema(json.dumps(ACTION_SCHEMA), verbose=False)
    r3 = run("List the action items as JSON. Each item has owner (a name or null) and text.", 500, grammar)
    try:
        json.loads(r3["text"])
        r3["json_ok"] = True
    except json.JSONDecodeError:
        r3["json_ok"] = False
    emit(check="json_task_same_prefix", **r3)
    emit(check="total", rss_mib_end=round(rss_mib() - before))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Text generation interface (spec B section 4.1)."""

from __future__ import annotations

import asyncio
from collections.abc import AsyncIterator, Callable
from dataclasses import dataclass
from typing import Any, Protocol

from forenote.scheduler import CancelToken

STALL_S = 120.0  # spec 10.2: no token for this time stops the generation


class LlmError(RuntimeError):
    """The model cannot run a request."""


class GenerationStalled(LlmError):
    """No token came for the stall time."""


class PromptTooLong(LlmError):
    """The prompt and the output do not fit the context."""


@dataclass(frozen=True)
class GenParams:
    max_tokens: int
    temperature: float = 0.2
    json_schema: dict[str, Any] | None = None


class TextGenerator(Protocol):
    model_id: str
    context_size: int

    def render(self, user_text: str) -> str:
        """Return the full prompt for one user message, with the chat template applied."""

    def count_tokens(self, prompt: str) -> int:
        """Count the tokens of a rendered prompt."""

    def generate(self, prompt: str, params: GenParams, cancel: CancelToken) -> AsyncIterator[str]:
        """Yield text pieces. Stop early when cancel is set."""


async def collect(
    gen: TextGenerator,
    prompt: str,
    params: GenParams,
    cancel: CancelToken,
    on_delta: Callable[[str], None],
    stall_s: float = STALL_S,
) -> str:
    """Run one generation. Call on_delta for each piece and return the full text.

    Raise GenerationStalled when no piece comes for stall_s seconds, and
    JobCancelled when cancel is set.
    """
    stream = gen.generate(prompt, params, cancel)
    parts: list[str] = []
    try:
        while True:
            try:
                piece = await asyncio.wait_for(anext(stream), stall_s)
            except StopAsyncIteration:
                break
            except TimeoutError:
                raise GenerationStalled(f"no token for {stall_s:.0f} s") from None
            parts.append(piece)
            on_delta(piece)
            if cancel.cancelled:
                break
    finally:
        aclose = getattr(stream, "aclose", None)
        if aclose is not None:
            await aclose()
    cancel.check()
    return "".join(parts)
```

import asyncio

class FakeTextGenerator:
    """Yield the scripted pieces. Record each prompt.

    outputs: a list of pieces for every call, or a callable prompt -> list of pieces.
    delay_s: {piece index: seconds to wait before that piece}.
    on_token: called with the piece index after each piece.
    error: raised after the last piece.
    """

    model_id = "fake-llm"
    context_size = 4096

    def __init__(
        self,
        outputs: list[str] | Callable[[str], list[str]],
        delay_s: dict[int, float] | None = None,
        on_token: Callable[[int], None] | None = None,
        error: BaseException | None = None,
    ):
        self._outputs = outputs
        self._delay_s = delay_s or {}
        self._on_token = on_token
        self._error = error
        self.prompts: list[str] = []
        self.params: list[GenParams] = []
        self.closed = False

    def render(self, user_text: str) -> str:
        return f"<user>\n{user_text}\n<model>\n"

    def count_tokens(self, prompt: str) -> int:
        return len(prompt.split())

    async def generate(self, prompt: str, params: GenParams, cancel: CancelToken) -> AsyncIterator[str]:
        self.prompts.append(prompt)
        self.params.append(params)
        pieces = self._outputs(prompt) if callable(self._outputs) else self._outputs
        try:
            for i, piece in enumerate(pieces):
                if cancel.cancelled:
                    return
                if i in self._delay_s:
                    await asyncio.sleep(self._delay_s[i])
                yield piece
                if self._on_token:
                    self._on_token(i)
            if self._error is not None:
                raise self._error
        finally:
            self.closed = True

```python
"""llama.cpp backend for TextGenerator (spec B section 4.1, decision 0003)."""

from __future__ import annotations

import asyncio
import contextlib
import json
import threading
from collections.abc import AsyncIterator
from pathlib import Path
from typing import Any

from forenote.llm.base import GenParams, LlmError, PromptTooLong
from forenote.scheduler import CancelToken

_DONE = object()


class LlamaCppGenerator:
    """One loaded GGUF model. Only one generation uses the model at a time.

    load() and close() block. Call them in a worker thread.
    """

    def __init__(self, model_path: Path, model_id: str, context_size: int, threads: int):
        self.model_path = model_path
        self.model_id = model_id
        self.context_size = context_size
        self.threads = threads
        self._llm: Any = None
        self._formatter: Any = None
        self._grammar_type: Any = None
        self._lock = threading.Lock()  # the model is not thread-safe

    @property
    def loaded(self) -> bool:
        return self._llm is not None

    def load(self) -> None:
        import llama_cpp
        from llama_cpp.llama_chat_format import Jinja2ChatFormatter

        llm = llama_cpp.Llama(
            model_path=str(self.model_path),
            n_ctx=self.context_size,
            n_threads=self.threads,
            n_threads_batch=self.threads,
            n_batch=512,
            verbose=False,
        )
        template = llm.metadata.get("tokenizer.chat_template")
        if not template:
            llm.close()
            raise LlmError("the model has no chat template")
        self._formatter = Jinja2ChatFormatter(
            template=template,
            eos_token=llm._model.token_get_text(llm.token_eos()),
            bos_token=llm._model.token_get_text(llm.token_bos()),
        )
        self._grammar_type = llama_cpp.LlamaGrammar
        self._llm = llm

    def close(self) -> None:
        llm, self._llm = self._llm, None
        if llm is not None:
            with self._lock:
                llm.close()

    def render(self, user_text: str) -> str:
        if self._formatter is None:
            raise LlmError("the model is not loaded")
        return self._formatter(messages=[{"role": "user", "content": user_text}]).prompt

    def _tokens(self, prompt: str) -> list[int]:
        # The rendered prompt already holds the BOS text (decision 0003).
        return self._llm.tokenize(prompt.encode("utf-8"), add_bos=False, special=True)

    def count_tokens(self, prompt: str) -> int:
        if self._llm is None:
            raise LlmError("the model is not loaded")
        return len(self._tokens(prompt))

    async def generate(self, prompt: str, params: GenParams, cancel: CancelToken) -> AsyncIterator[str]:
        if self._llm is None:
            raise LlmError("the model is not loaded")
        loop = asyncio.get_running_loop()
        queue: asyncio.Queue[Any] = asyncio.Queue()
        stop = threading.Event()

        def put(item: Any) -> None:
            with contextlib.suppress(RuntimeError):  # the loop can be closed already
                loop.call_soon_threadsafe(queue.put_nowait, item)

        def work() -> None:
            try:
                with self._lock:
                    llm = self._llm
                    if llm is None:
                        raise LlmError("the model is not loaded")
                    tokens = self._tokens(prompt)
                    if len(tokens) + params.max_tokens > self.context_size:
                        raise PromptTooLong(
                            f"{len(tokens)} prompt tokens + {params.max_tokens} output tokens "
                            f"> context {self.context_size}"
                        )
                    grammar = None
                    if params.json_schema is not None:
                        grammar = self._grammar_type.from_json_schema(json.dumps(params.json_schema), verbose=False)
                    for chunk in llm.create_completion(
                        tokens,
                        max_tokens=params.max_tokens,
                        temperature=params.temperature,
                        stream=True,
                        grammar=grammar,
                    ):
                        if stop.is_set() or cancel.cancelled:
                            break
                        text = chunk["choices"][0]["text"]
                        if text:
                            put(text)
            except BaseException as e:  # noqa: BLE001 - hand every error to the loop
                put(e)
            else:
                put(_DONE)

        threading.Thread(target=work, daemon=True, name="forenote-llm").start()
        try:
            while True:
                item = await queue.get()
                if item is _DONE:
                    return
                if isinstance(item, BaseException):
                    raise item
                yield item
        finally:
            stop.set()
```

Note: the worker thread keeps the lock until `create_completion` returns. After a stall, the next `generate` waits for the old thread in its own thread, not on the event loop.

```python
"""Bullets and notes outputs (spec B sections 4.3 and 5)."""

from __future__ import annotations

import json
import sqlite3
from dataclasses import dataclass
from typing import Any

from forenote.store.jobs import now_iso

_BULLET_COLS = "id, meeting_id, position, text, t_written_ms, enhanced_text, enhanced_edited"


@dataclass(frozen=True)
class Bullet:
    id: int
    meeting_id: int
    position: int
    text: str
    t_written_ms: int | None
    enhanced_text: str | None
    enhanced_edited: bool

    def to_dict(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "position": self.position,
            "text": self.text,
            "t_written_ms": self.t_written_ms,
            "enhanced_text": self.enhanced_text,
            "enhanced_edited": self.enhanced_edited,
        }


def _bullet(row: tuple) -> Bullet:
    return Bullet(row[0], row[1], row[2], row[3], row[4], row[5], bool(row[6]))


class NotesStore:
    def __init__(self, conn: sqlite3.Connection):
        self._conn = conn

    def bullets(self, meeting_id: int) -> list[Bullet]:
        rows = self._conn.execute(
            f"SELECT {_BULLET_COLS} FROM bullets WHERE meeting_id = ? ORDER BY position", (meeting_id,)
        ).fetchall()
        return [_bullet(r) for r in rows]

    def bullet(self, bullet_id: int) -> Bullet | None:
        row = self._conn.execute(f"SELECT {_BULLET_COLS} FROM bullets WHERE id = ?", (bullet_id,)).fetchone()
        return _bullet(row) if row else None

    def add_bullet(self, meeting_id: int, text: str, t_written_ms: int | None) -> Bullet:
        with self._conn:
            (last,) = self._conn.execute(
                "SELECT COALESCE(MAX(position), 0) FROM bullets WHERE meeting_id = ?", (meeting_id,)
            ).fetchone()
            cur = self._conn.execute(
                "INSERT INTO bullets(meeting_id, position, text, t_written_ms) VALUES (?, ?, ?, ?)",
                (meeting_id, last + 1, text, t_written_ms),
            )
        return self.bullet(int(cur.lastrowid))

    def replace_bullets(
        self, meeting_id: int, items: list[tuple[int | None, str]], t_new_ms: int | None
    ) -> list[Bullet]:
        """Make the list of the meeting equal to items. Known ids keep time and expansion."""
        known = {b.id for b in self.bullets(meeting_id)}
        for bullet_id, _ in items:
            if bullet_id is not None and bullet_id not in known:
                raise ValueError(f"bullet {bullet_id} is not in meeting {meeting_id}")
        keep = {bullet_id for bullet_id, _ in items if bullet_id is not None}
        with self._conn:
            for bullet_id in known - keep:
                self._conn.execute("DELETE FROM bullets WHERE id = ?", (bullet_id,))
            for position, (bullet_id, text) in enumerate(items, 1):
                if bullet_id is None:
                    self._conn.execute(
                        "INSERT INTO bullets(meeting_id, position, text, t_written_ms) VALUES (?, ?, ?, ?)",
                        (meeting_id, position, text, t_new_ms),
                    )
                else:
                    self._conn.execute(
                        "UPDATE bullets SET position = ?, text = ? WHERE id = ?", (position, text, bullet_id)
                    )
        return self.bullets(meeting_id)

    def set_enhanced(self, bullet_id: int, text: str | None, edited: bool) -> bool:
        with self._conn:
            cur = self._conn.execute(
                "UPDATE bullets SET enhanced_text = ?, enhanced_edited = ? WHERE id = ?",
                (text, int(edited), bullet_id),
            )
        return cur.rowcount == 1

    def output(self, meeting_id: int) -> dict[str, Any] | None:
        row = self._conn.execute(
            "SELECT summary, action_items, model, mode, created_at FROM meeting_outputs WHERE meeting_id = ?",
            (meeting_id,),
        ).fetchone()
        if row is None:
            return None
        return {
            "summary": row[0],
            "action_items": json.loads(row[1]) if row[1] is not None else None,
            "model": row[2],
            "mode": row[3],
            "created_at": row[4],
        }

    def start_output(self, meeting_id: int, model: str, mode: str) -> None:
        """Start a new result: clear the summary and the action items."""
        with self._conn:
            self._conn.execute(
                "INSERT INTO meeting_outputs(meeting_id, summary, action_items, model, mode, created_at) "
                "VALUES (?, NULL, NULL, ?, ?, ?) "
                "ON CONFLICT(meeting_id) DO UPDATE SET summary = NULL, action_items = NULL, "
                "model = excluded.model, mode = excluded.mode, created_at = excluded.created_at",
                (meeting_id, model, mode, now_iso()),
            )

    def set_summary(self, meeting_id: int, text: str) -> None:
        with self._conn:
            self._conn.execute("UPDATE meeting_outputs SET summary = ? WHERE meeting_id = ?", (text, meeting_id))

    def set_action_items(self, meeting_id: int, items: list[dict[str, Any]]) -> None:
        with self._conn:
            self._conn.execute(
                "UPDATE meeting_outputs SET action_items = ? WHERE meeting_id = ?", (json.dumps(items), meeting_id)
            )
```
